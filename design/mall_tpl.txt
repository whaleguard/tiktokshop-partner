                ${esc(order.carrier || '')}
              </dd>

            </div>

            <div class="info-row">

              <dt>
                ${esc(t('m_tracking_number'))}
              </dt>

              <dd class="bank-account">

                <span>
                  ${esc(order.trackingNo)}
                </span>

                <button
                  type="button"
                  class="copy-button"
                  data-action="copy-tracking"
                  data-value="${esc(order.trackingNo)}">

                  ${esc(t('m_copy'))}

                </button>

              </dd>

            </div>

          </dl>


          ${order.trackingUrl ? `
            <a
              class="btn btn-outline btn-block"
              style="margin-top:12px"
              href="${esc(order.trackingUrl)}"
              target="_blank"
              rel="noopener noreferrer">

              ${esc(t('m_track_delivery'))}

            </a>
          ` : ''}

        </div>

      ` : ''}


      <!-- ORDER ITEM LIST -->
      <section>

        <h3 class="block-title">
          ${esc(t('m_order_products'))}
        </h3>

        ${(order.items || [])
          .map(item => MALL_UI.orderItem(item))
          .join('')}

      </section>

      <hr class="m-divider">

      ${MALL_UI.moneySummary({
        subtotal: order.subtotal,
        shipping: order.shipping,
        discount: order.discount,
        total: order.total
      })}

    </article>
  `,


  /* ------------------------------------------------
     ORDER STATUS TRACKER
  ------------------------------------------------ */

  orderStatus: (status) => `

    <ol class="order-status"
        aria-label="${esc(t('m_order_status'))}">

      ${[
        'received',
        'paid',
        'preparing',
        'shipped',
        'delivered'
      ].map((step, i, steps) => {

        const current = steps.indexOf(status);

        return `
          <li
            class="status-step
              ${i < current ? 'done' : ''}
              ${i === current ? 'current' : ''}"
            ${i === current ? 'aria-current="step"' : ''}>

            <span class="status-dot"
                  aria-hidden="true">

              ${i < current ? '✓' : i + 1}

            </span>

            <span class="status-label">
              ${esc(t(`m_status_${step}`))}
            </span>

          </li>
        `;

      }).join('')}

    </ol>
  `
